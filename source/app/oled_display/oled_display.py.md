_I2C3_PADS = ((0x100C0010, 0x1002), (0x100C0014, 0x1002))


def set_pad_func(phys_addr, func_val):
    """mmap /dev/mem 写 iocfg 寄存器,把 pad 切到目标功能。"""
    PAGE = 0x1000
    fd = os.open("/dev/mem", os.O_RDWR)
    try:
        m = mmap.mmap(fd, PAGE, mmap.MAP_SHARED,
                      mmap.PROT_READ | mmap.PROT_WRITE,
                      offset=phys_addr & ~(PAGE - 1))
        off = phys_addr & (PAGE - 1)
        m[off:off + 4] = struct.pack('<I', func_val & 0xFFFFFFFF)
        m.close()
    finally:
        os.close(fd)

# SSD1306 128x64 标准初始化序列(0.96"/1.3" 模组通用;0x8D,0x14 电荷泵必须)
_INIT = bytes([
    0xAE, 0xD5, 0x80, 0xA8, 0x3F, 0xD3, 0x00, 0x40, 0x8D, 0x14,
    0x20, 0x02, 0xA1, 0xC8, 0xDA, 0x12, 0x81, 0xCF, 0xD9, 0xF1,
    0xDB, 0x40, 0xA4, 0xA6, 0xAF,
])


class SSD1306:
    """128x64 OLED 驱动(/dev/i2c-N + I2C_SLAVE + os.write,只写)。

    显存页寻址:gram[(y>>3)*128 + x] 的 bit(y&7),bit0=最上一行。
    所有绘图只改内存帧缓冲,flush() 一次性整屏刷新。
    """

    def __init__(self, bus=3, addr=0x3C):
        self.W, self.H, self.PAGES = 128, 64, 8
        self._fd = os.open(f"/dev/i2c-{bus}", os.O_RDWR)
        fcntl.ioctl(self._fd, I2C_SLAVE, addr)
        self._gram = bytearray(self.W * self.PAGES)
        self._cmd(_INIT)
        self.clear()
        self.flush()

    def _cmd(self, cmds):
        os.write(self._fd, b"\x00" + bytes(cmds))   # 控制字节 0x00:后续全命令

    def _data(self, data):
        os.write(self._fd, b"\x40" + bytes(data))   # 控制字节 0x40:后续全数据

    def on(self, v=True):
        self._cmd(b"\xAF" if v else b"\xAE")

    def clear(self):
        self._gram[:] = b"\x00" * len(self._gram)

    def pixel(self, x, y, on=True):
        if 0 <= x < self.W and 0 <= y < self.H:
            idx = (y >> 3) * self.W + x
            if on:
                self._gram[idx] |= 1 << (y & 7)
            else:
                self._gram[idx] &= ~(1 << (y & 7)) & 0xFF

    def char(self, x, y, ch):
        o = (ord(ch) - 0x20) * 16 if 0x20 <= ord(ch) <= 0x7E else 0
        for r in range(16):
            line = _FONT[o + r]
            for c in range(8):
                if line & (0x80 >> c):
                    self.pixel(x + c, y + r, True)

    def text(self, x, y, s):
        cx, cy = x, y
        for ch in s:
            if ch == "\n" or cx + 8 > self.W:
                cx, cy = x, cy + 16
                if ch == "\n":
                    continue
            self.char(cx, cy, ch)
            cx += 8

    def flush(self):
        for p in range(self.PAGES):
            self._cmd(bytes([0xB0 + p, 0x10, 0x02]))   # 页地址 / 列高0 / 列低2(SH1106偏移)
            self._data(self._gram[p * self.W:(p + 1) * self.W])

    def close(self):
        if self._fd >= 0:
            os.close(self._fd)
            self._fd = -1

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.close()


def main():
    for pad, func in _I2C3_PADS:
        set_pad_func(pad, func)
    with SSD1306() as oled:
        print("OLED 每 1s 刷新 uptime,Ctrl+C 退出", flush=True)
        try:
            while True:
                up = float(open("/proc/uptime").read().split()[0])
                oled.clear()
                oled.text(0, 0, "OLED  Python")
                oled.text(0, 24, f"UP {up:.0f}s")
                oled.flush()
                time.sleep(1)
        except KeyboardInterrupt:
            oled.clear()
            oled.flush()
            oled.on(False)
            print("\n退出", flush=True)


if __name__ == "__main__":
    main()
